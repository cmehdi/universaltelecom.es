Send a Simple SMS

const axios = require('axios');

const sendSMS = async () => {
  const response = await axios.post('https://api.universaltelecom.com/v1/sms', {
    from: 'YourBrand',
    to: '+34612345678',
    message: 'Hello! This is a test message from Universal Telecom SMS API.'
  }, {
    headers: {
      'Authorization': 'Bearer YOUR_API_KEY',
      'Content-Type': 'application/json'
    }
  });
  
  console.log('SMS sent successfully!');
  console.log('Message ID:', response.data.message_id);
  console.log('Status:', response.data.status);
  return response.data;
};

import requests

def send_sms():
    url = 'https://api.universaltelecom.com/v1/sms'
    headers = {
        'Authorization': 'Bearer YOUR_API_KEY',
        'Content-Type': 'application/json'
    }
    data = {
        'from': 'YourBrand',
        'to': '+34612345678',
        'message': 'Hello! This is a test message from Universal Telecom SMS API.'
    }
    
    response = requests.post(url, json=data, headers=headers)
    result = response.json()
    
    print('SMS sent successfully!')
    print(f"Message ID: {result['message_id']}")
    print(f"Status: {result['status']}")
    return result

Send Bulk SMS

const axios = require('axios');

const sendBulkSMS = async () => {
  const recipients = [
    { to: '+34612345678', message: 'Hello John!' },
    { to: '+34612345679', message: 'Hello Sarah!' },
    { to: '+34612345680', message: 'Hello Mike!' }
  ];

  const response = await axios.post('https://api.universaltelecom.com/v1/sms/bulk', {
    from: 'YourBrand',
    messages: recipients
  }, {
    headers: {
      'Authorization': 'Bearer YOUR_API_KEY',
      'Content-Type': 'application/json'
    }
  });
  
  console.log(`Successfully sent ${response.data.sent_count} messages`);
  console.log('Batch ID:', response.data.batch_id);
  return response.data;
};

Schedule SMS for Later

const axios = require('axios');

const scheduleSMS = async () => {
  // Schedule for tomorrow at 9 AM
  const scheduledTime = new Date();
  scheduledTime.setDate(scheduledTime.getDate() + 1);
  scheduledTime.setHours(9, 0, 0, 0);

  const response = await axios.post('https://api.universaltelecom.com/v1/sms', {
    from: 'YourBrand',
    to: '+34612345678',
    message: 'This is your scheduled reminder!',
    scheduled_at: scheduledTime.toISOString()
  }, {
    headers: {
      'Authorization': 'Bearer YOUR_API_KEY',
      'Content-Type': 'application/json'
    }
  });
  
  console.log('SMS scheduled successfully!');
  console.log('Will be sent at:', scheduledTime);
  return response.data;
};

scheduleSMS();
import requests
from datetime import datetime, timedelta

def schedule_sms():
    url = 'https://api.universaltelecom.com/v1/sms'
    headers = {
        'Authorization': 'Bearer YOUR_API_KEY',
        'Content-Type': 'application/json'
    }
    
    # Schedule for tomorrow at 9 AM
    scheduled_time = datetime.now() + timedelta(days=1)
    scheduled_time = scheduled_time.replace(hour=9, minute=0, second=0)
    
    data = {
        'from': 'YourBrand',
        'to': '+34612345678',
        'message': 'This is your scheduled reminder!',
        'scheduled_at': scheduled_time.isoformat()
    }
    
    response = requests.post(url, json=data, headers=headers)
    result = response.json()
    
    print('SMS scheduled successfully!')
    print(f'Will be sent at: {scheduled_time}')
    return result

Frequently Asked Questions

How quickly are SMS messages delivered?

Our average delivery time is under 3 seconds globally. For high-priority markets like Spain, France, and Germany, delivery typically occurs in under 2 seconds.

What is your delivery success rate?

We maintain a 99.9% delivery success rate thanks to our direct carrier connections and intelligent routing system that automatically switches to backup routes if needed.

Can I send SMS to any country?

Yes! We support SMS delivery to over 190 countries worldwide. Check our pricing page for specific country rates and availability.

Do you support two-way SMS messaging?

You can receive SMS replies by configuring a webhook URL in your dashboard. We'll POST incoming messages to your endpoint in real-time.

What is the maximum length for an SMS?

Standard SMS messages are limited to 160 characters. Longer messages are automatically split into multiple parts (concatenated SMS), with each part charged separately.

Can I use my own sender ID?

Yes, in countries where alphanumeric sender IDs are supported. Some countries require numeric sender IDs or have specific regulations. Contact us for country-specific details.

How do volume discounts work?

Volume discounts are automatically applied based on your monthly usage: 5% off at 10K SMS, 10% off at 50K SMS, 15% off at 100K SMS, and 20% off at 500K+ SMS per month.

Is there a free trial available?

Yes! Sign up for a free account and receive €10 in credits to test our SMS API. No credit card required for the trial period.

What happens if a message fails to deliver?

You won't be charged for failed messages. You'll receive a delivery report via webhook or API with the failure reason, allowing you to retry or take appropriate action.

Do you provide SDKs?

Yes! We offer official SDKs for Node.js, Python, PHP, Ruby, Java, C#, and Go. All SDKs are open-source and available on GitHub.

How secure is your SMS API?

All API communications use TLS 1.3 encryption. We support IP whitelisting, OAuth 2.0 authentication, and are SOC 2 Type II certified.

Can I schedule SMS messages?

Yes! You can schedule messages for future delivery by specifying a timestamp in your API request. Perfect for time-sensitive campaigns or reminders.
