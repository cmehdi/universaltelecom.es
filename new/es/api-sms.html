Integración Sencilla

Envía tu primer SMS con solo unas líneas de código

Envío Simple de SMS

const UniversalTelecom = require('universal-telecom');

const client = new UniversalTelecom({
  apiKey: 'tu_api_key',
  apiSecret: 'tu_api_secret'
});

// Enviar SMS simple
const message = await client.sms.send({
  from: 'TuEmpresa', // O un número: '+34900123456'
  to: '+34612345678',
  body: '¡Hola! Tu código de verificación es: 123456'
});

console.log('Message SID:', message.sid);
console.log('Status:', message.status);
console.log('Cost:', message.price);

SMS Programado

from datetime import datetime, timedelta

# Programar envío para dentro de 2 horas

message = client.sms.send(
    from_='TuEmpresa',
    to='+34612345678',
    body='Recordatorio: Tu cita es mañana a las 10:00',
    send_at=send_at
)

print(f'Mensaje programado para: {send_at}')

Recibir SMS (Webhook)

<?php
// webhook.php - Recibir SMS entrantes

// Verificar firma de seguridad
$signature = $_SERVER['HTTP_X_UNIVERSAL_SIGNATURE'];
$isValid = $client->validateWebhook($signature, file_get_contents('php://input'));

if ($isValid) {
    $data = json_decode(file_get_contents('php://input'), true);
    
    $from = $data['from'];
    $body = $data['body'];
    $messageId = $data['message_id'];
    
    // Procesar el mensaje recibido
    echo "Mensaje de $from: $body";
    
    // Responder automáticamente
    $client->sms->send([
        'from' => '+34900123456',
        'to' => $from,
        'body' => 'Gracias por tu mensaje. Te responderemos pronto.'
    ]);
}

Envío Simple de SMS

curl -X POST https://api.universaltelecom.com/v1/sms \
  -u tu_api_key:tu_api_secret \
  -H "Content-Type: application/json" \
  -d '{
    "from": "TuEmpresa",
    "to": "+34612345678",
    "body": "¡Hola! Tu código de verificación es: 123456"
  }'

Consultar Estado de SMS

curl -X GET https://api.universaltelecom.com/v1/sms/SM123456 \
  -u tu_api_key:tu_api_secret

Obtener Historial de SMS

curl -X GET "https://api.universaltelecom.com/v1/sms?limit=50&status=delivered" \
  -u tu_api_key:tu_api_secret
