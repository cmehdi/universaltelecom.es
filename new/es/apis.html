Ejemplos de Código

Comienza a integrar en minutos con nuestros SDKs

Realizar una Llamada Telefónica

const UniversalTelecom = require('universal-telecom');

const client = new UniversalTelecom({
  apiKey: 'tu_api_key',
  apiSecret: 'tu_api_secret'
});

// Realizar una llamada
const call = await client.voice.createCall({
  from: '+34900123456',
  to: '+34612345678',
  url: 'https://tuapp.com/ivr-config.xml'
});

console.log('Call SID:', call.sid);
console.log('Status:', call.status);

Enviar un SMS

// Enviar SMS
const message = await client.sms.send({
  from: '+34900123456',
  to: '+34612345678',
  body: '¡Hola! Este es un mensaje de prueba.'
});

console.log('Message SID:', message.sid);
console.log('Status:', message.status);

Realizar una Llamada Telefónica

from universal_telecom import Client

client = Client(
    api_key='tu_api_key',
    api_secret='tu_api_secret'
)

# Realizar una llamada
call = client.voice.create_call(
    from_='+34900123456',
    to='+34612345678',
    url='https://tuapp.com/ivr-config.xml'
)

print(f'Call SID: {call.sid}')
print(f'Status: {call.status}')

Enviar un SMS

# Enviar SMS
message = client.sms.send(
    from_='+34900123456',
    to='+34612345678',
    body='¡Hola! Este es un mensaje de prueba.'
)

print(f'Message SID: {message.sid}')
print(f'Status: {message.status}')

Realizar una Llamada Telefónica

<?php
require_once 'vendor/autoload.php';

use UniversalTelecom\Client;

$client = new Client(
    'tu_api_key',
    'tu_api_secret'
);

// Realizar una llamada
$call = $client->voice->createCall([
    'from' => '+34900123456',
    'to' => '+34612345678',
    'url' => 'https://tuapp.com/ivr-config.xml'
]);

echo 'Call SID: ' . $call->sid;
echo 'Status: ' . $call->status;

Enviar un SMS

// Enviar SMS
$message = $client->sms->send([
    'from' => '+34900123456',
    'to' => '+34612345678',
    'body' => '¡Hola! Este es un mensaje de prueba.'
]);

echo 'Message SID: ' . $message->sid;
echo 'Status: ' . $message->status;

Realizar una Llamada Telefónica

curl -X POST https://api.universaltelecom.com/v1/calls \
  -u tu_api_key:tu_api_secret \
  -d "From=+34900123456" \
  -d "To=+34612345678" \
  -d "Url=https://tuapp.com/ivr-config.xml"

Enviar un SMS

curl -X POST https://api.universaltelecom.com/v1/sms \
  -u tu_api_key:tu_api_secret \
  -d "From=+34900123456" \
  -d "To=+34612345678" \
  -d "Body=¡Hola! Este es un mensaje de prueba."
